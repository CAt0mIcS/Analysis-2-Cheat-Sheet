\subsection{Parameterabhängige Integrale}
\begin{sectionbox}
    \subsubsection{Stetigkeit}
    Sei $I=[a,b], J=[c,d], f:I \times J \rightarrow \mathbb R$ auf $I \times J$ stetig
    \begin{equation*}
        \Longrightarrow F(x)=\int_c^d f(x,y)\mathrm dy \quad \text{stetig auf } I
    \end{equation*}
\end{sectionbox}

\begin{sectionbox}
    \subsubsection{Differenzierbarkeit}
    Sei $I=[a,b], J=[c,d], f:I \times J \rightarrow \mathbb R$ und $\partial_x f$ auf $I \times J$ stetig
    \begin{equation*}
        \Longrightarrow F(x)=\int_{c(x)}^{d(x)} f(x,y)\mathrm dy \quad \text{stetig diffb. auf } I
    \end{equation*}
    und
    \begin{equation*}
        \resizebox{\hsize}{!}
        {
            $F'(x)=\int_{c(x)}^{d(x)} f(x,y)\mathrm dy + f(x, d(x))d'(x) - f(x, c(x))c'(x)$
        }
    \end{equation*}
    Weiterhin gilt:
    \begin{equation*}
        F'(x)=\partial_x \int_{c}^{d} f(x,y)\mathrm dy = \int_{c}^{d}\partial_x f(x,y) \mathrm dy
    \end{equation*}
\end{sectionbox}


\subsection{Bereichsintegrale}
\begin{sectionbox}
    \subsubsection{Fubini für Bereichsintegrale}
    $I=[a,b], J=[c,d], K=[e,f], f: I \times J \times K \rightarrow \mathbb R$ stetig auf $I \times J \times K$
    \begin{equation*}
        \resizebox{\hsize}{!}
        {
            $\Longrightarrow \int_a^b\int_c^d\int_e^f f(x,y,z)\mathrm dz \mathrm dy \mathrm dx = \int_e^f\int_c^d\int_a^b f(x,y,z)\mathrm dx \mathrm dy \mathrm dz$
        }
    \end{equation*}
\end{sectionbox}

\begin{sectionbox}
    \subsubsection{Krummlinige Koordinaten - Transformationssatz}
    $\Omega \subset \mathbb R^n$ offen und $\Phi : \Omega \rightarrow \mathbb R^n$ reguläre, stetig diffb. Transformation. $\det J_\Phi\ne 0$ außer für Nullmengen
    \begin{equation*}
        \int_{\Phi(\Omega)} f(x)=\int_\Omega f(\Phi(u))|\det{J_\Phi(u)}|\mathrm du
    \end{equation*}
\end{sectionbox}

\begin{sectionbox}
    \subsubsection{Normalbereiche}
    \begin{enumerate}
        \item Typ 1: $\iint_\Omega f(x,y)\mathrm d(x,y)=\int_a^b\int_{c(x)}^{d(x)}f(x,y)\mathrm dy \mathrm dx$
        \item Typ 2: $\iint_\Omega f(x,y)\mathrm d(x,y)=\int_c^d\int_{a(y)}^{b(y)}f(x,y)\mathrm dx \mathrm dy$
    \end{enumerate}
\end{sectionbox}

\subsection{Oberflächenintegral}
\begin{sectionbox}
    $B \subset \mathbb R^2$ Jordan-messbar, $\varphi: B \rightarrow \mathbb R^3$ stückweise stetig diffbar beschreibt eine Fläche im $\mathbb R^3$
    \begin{equation*}
        \varphi(u,v)=\begin{pmatrix}
            x(u,v) \\ y(u,v) \\ z(u,v)
        \end{pmatrix}
    \end{equation*}

    \begin{description}
        \item[regulär] Falls alle Punkte regulär $\rightarrow$ alle Punkte $\underbrace{\partial_u\varphi(u_0,v_0) \times \partial_v\varphi(u_0,v_0)}_{\text{Normalenvektor (nicht normiert)}} \ne 0$
    \end{description}
\end{sectionbox}

\begin{sectionbox}
    \subsubsection{Flächenintegrale für \textcolor{green}{\text{Skalarfelder}}/\textcolor{red}{\text{Vektorfelder}}}
    Sei $f : \mathbb R^3 \rightarrow \mathbb R^{\textcolor{red}{3}}$ ein beschränktes \textcolor{green}{\text{Skalarfelder}} / \textcolor{red}{\text{Vektorfelder}}, $\varphi : B \rightarrow \mathbb R^3$ eine Fläche
    \begin{equation*}
        \resizebox{\hsize}{!}
        {
            $\iint_\varphi f \mathrm dA=\iint_B f(\varphi(u,v)) (\textcolor{green}{||} \partial_u \varphi(u,v) \times \partial_v \varphi(u,v) \textcolor{green}{||})\mathrm d(u,v)$
        }
    \end{equation*}

    $\textcolor{red}{\iint_\varphi f \cdot \mathrm dA = \iint_\varphi (f \cdot n)\mathrm dA}$
\end{sectionbox}

\subsection{Integralsätze}
Parametrisierung von $\delta \Omega$ (geschlossene, stückweise reguläre Kurve) immer so wählen, dass $\Omega \subset \mathbb R^2$ immer links zur Durchlaufrichtung ist
\begin{emphbox}
    \subsubsection{Satz von Green}
    Sei $\Omega \subset \mathbb R^2$ mit Rand $\delta \Omega$ wie oben. Sei $v: \overline{\Omega} \rightarrow \mathbb R^2$ stetig diffb.
    \begin{equation*}
        \iint_{\Omega}\partial_xv_2-\partial_yv_1 \mathrm d(x,y)=\int_{\delta\Omega}v \cdot \mathrm d(x,y)
    \end{equation*}
\end{emphbox}

\begin{emphbox}
    \subsubsection{Satz von Gauß}
    Sei $\Omega \subset \mathbb R^3$ und sei $\delta \Omega$ durch eine stückweise reguläre, geschlossene Fläche parametrisiert. 
    Sei $v : \overline{\Omega} \rightarrow \mathbb R^3$ stetig diffb.
    \begin{equation*}
        \iiint_\Omega \div (v) \mathrm d(x,y,z) = \iint_{\delta \Omega}v^Tn\mathrm dA=\iint_{\delta \Omega}v \cdot \mathrm dA
    \end{equation*}
    mit $n$ Einheitsnormalenvektor der Fläche, der nach außen zeigt.
\end{emphbox}

\begin{emphbox}
    \subsubsection{Greensche Formel}
    Sei $\Omega \subset \mathbb R^3$ und $\delta\Omega$ wie oben. Sei $\varphi,u : \overline{\Omega}\rightarrow \mathbb R$ stetig diffb. bzw. 2x stetig diffb.
    \begin{equation*}
        \resizebox{\hsize}{!}
        {
            $\iiint_\Omega \Delta u(x) \varphi(x)\mathrm dx=\iint_{\delta \Omega}\varphi(x)\nabla u^T(x)n\mathrm dA - \iiint_\Omega \nabla u(x)\nabla \varphi(x)\mathrm dx$
        }
    \end{equation*}
\end{emphbox}
