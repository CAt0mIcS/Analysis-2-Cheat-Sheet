Vektorfeld $f: \Omega \subset \mathbb R^n \rightarrow \mathbb R^m$ besteht aus $m$ Komponentenfunktionen $f_i : \Omega \rightarrow \mathbb R$ (Skalarfelder).

\subsection{Definitionen}
\begin{description}
    \item[stetig in $x_0$] Alle $f_i$ stetig in $x_0$
    \item[partiell diffbar in $x_0 \in \Omega$] Alle $f_i$ partiell diffbar in $x_0$
    \item[stetig diffbar] Alle $f_i$ stetig diffbar
\end{description}

\subsection{Jacobi-Matrix}
Für partiell diffbares Vektorfeld gilt:

\begin{equation*}
    \resizebox{.98\hsize}{!}{
        $J_f(x)=\begin{pmatrix}
            \partial x_1 f_1(x) & \dots  & \partial x_nf_1(x) \\
            \partial x_1 f_2(x) & \dots  & \partial x_nf_2(x) \\
            \vdots              & \ddots & \vdots             \\
            \partial x_1 f_m(x) & \dots  & \partial x_nf_m(x) \\
        \end{pmatrix} = \begin{pmatrix}
            \nabla f_1(x)^\top \\
            \nabla f_2(x)^\top \\
            \vdots             \\
            \nabla f_m(x)^\top
        \end{pmatrix}$
    }
\end{equation*}

\subsubsection{Gradientenfeld}
Der Gradient eines Skalarfeldes ist ein Vektorfeld. \textbf{Es gilt}: \\
\hspace*{.1\textwidth}$J_{\nabla g}(x)=H_g(x)$

\subsection{Fréchet-Diffbarkeit $Df(x)$ am festen Punkt $x$}
\begin{equation*}
    \begin{split}
         & f(x+h)=f(x)+Df(x)(h) + o(||h||) \\
         & Df(x)(h)=J_f(x)h
    \end{split}
\end{equation*}


$Df(x)(h)$ ist hier eine linerare Abbildung (Approximation) am Punkt $x$ mit der Variable $h$

\subsection{Rechenregeln}
\subsubsection{Produktregel}
$f,g \in \Omega \subset \mathbb R^n \rightarrow \mathbb R^m$ partiell diffbar, $h(x) = f(x)^\top g(x)$ auch partiell diffbar und es gilt:
\begin{equation*}
    J_h(x)=g(x)^\top J_f(x)+f(x)^\top J_g(x)
\end{equation*}

\subsubsection{Kettenregel}
$f: \Omega \rightarrow \mathbb R^n, g: D \rightarrow \mathbb R^m, h=g \circ f$
\begin{equation*}
    Dh(x)=Dg(f(x))\circ Df(x) \qquad J_h(x)=J_g(f(x))J_f(x)
\end{equation*}
